Thank you for installing {{ .Chart.Name }}!

Your release is named {{ .Release.Name }}.

To learn more about the release, try:

  $ helm status {{ .Release.Name }}
  $ helm get all {{ .Release.Name }}

1. Get the application URL by running these commands:
{{- if .Values.ingress.enabled }}
{{- range $host := .Values.ingress.hosts }}
  {{- range .paths }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}
{{- end }}
{{- else if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ include "api-gateway.namespace" . }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ include "api-gateway.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ include "api-gateway.namespace" . }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}
     NOTE: It may take a few minutes for the LoadBalancer IP to be available.
           You can watch the status by running 'kubectl get --namespace {{ include "api-gateway.namespace" . }} svc -w {{ include "api-gateway.fullname" . }}'
  export SERVICE_IP=$(kubectl get svc --namespace {{ include "api-gateway.namespace" . }} {{ include "api-gateway.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.service.port }}
{{- else }}
  export POD_NAME=$(kubectl get pods --namespace {{ include "api-gateway.namespace" . }} -l "app.kubernetes.io/name={{ include "api-gateway.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  export CONTAINER_PORT=$(kubectl get pod --namespace {{ include "api-gateway.namespace" . }} $POD_NAME -o jsonpath="{.spec.containers[0].ports[0].containerPort}")
  echo "Visit http://127.0.0.1:8080 to use your application"
  kubectl --namespace {{ include "api-gateway.namespace" . }} port-forward $POD_NAME 8080:$CONTAINER_PORT
{{- end }}

2. View the Gateway routes:
   kubectl get --namespace {{ include "api-gateway.namespace" . }} svc -l app.kubernetes.io/name={{ include "api-gateway.name" . }}

3. Check the health status:
   kubectl get pods -n {{ include "api-gateway.namespace" . }} -l app.kubernetes.io/name={{ include "api-gateway.name" . }}

4. View logs:
   kubectl logs -n {{ include "api-gateway.namespace" . }} -l app.kubernetes.io/name={{ include "api-gateway.name" . }} --tail=100 -f

5. Gateway Actuator Endpoints:
   - Health: http://localhost:8080/actuator/health
   - Gateway Routes: http://localhost:8080/actuator/gateway/routes
   - Metrics: http://localhost:8080/actuator/metrics
