interface CountryCardProps {
  country: {
    code: string;      // 'NG', 'KE', etc.

  };
  metrics: {
    revenue: number;

    dealsClosed: number;
    pipelineValue: number;
    winRate: number;
    trend: 'up' | 'down' | 'neutral';
  };
  onClick?: () => void;
}

// Usage Example
<CountryCard

  }}

/>

interface ComparisonTableProps<T> {
  data: T[];
  columns: ComparisonColumn<T>[];
  comparableMetric?: keyof T;
  showTrend?: boolean;
  showRanking?: boolean;
}

  key: keyof T;
  label: string;
  format: (value: any) => string;
